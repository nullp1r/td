17pub fn to_native(name: &str) -> Option<&'static str> {
18  match name {
19    "Int32" | "int32" => Some("i32"),
20    "Int53" | "int53" => Some("i64"),
21    "Int64" | "int64" => Some("i64"),
22    "Double" | "double" => Some("f64"),
23    "String" | "string" => Some("String"),
24    "Vector" | "vector" => Some("Vec<_>"),
25    "Bytes" | "bytes" => Some("Vec<u8>"),
26    "Bool" | "bool" => Some("bool"),
27    _ => None,
28  }
29}

31pub fn escaped_keyword(s: &str) -> impl fmt::Display {
32  fmt::from_fn(move |f| {
33    if RUST_KEYWORDS.contains(&s) {
34      f.write_str("r#")?;
35    }
36    f.write_str(s)
37  })
