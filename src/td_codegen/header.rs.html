1//! Source header formatting for generated Rust files.

3use std::fmt;
4use std::time::{Duration, UNIX_EPOCH};

6/// Formats a source header with schema provenance and generation metrics.
7pub fn header(schema: &str, [parse, codegen]: [Duration; 2]) -> impl fmt::Display + '_ {
8  fmt::from_fn(move |f| {
9    let mut lines = schema.lines();
10    let (Some(line1), Some(line2)) = (lines.next(), lines.next()) else { return Ok(()) };
11    if !line1.starts_with("// TDLib") || !line2.starts_with("// fetched") {
12      return Ok(());
13    }
14
15    let total = fmt_dur(parse + codegen);
16    let parse = fmt_dur(parse);
17    let codegen = fmt_dur(codegen);
18    let utc = fmt_utc();
19
20    writeln!(f, "{line1}")?;
21    writeln!(f, "{line2} · generated {utc} in {total} ({parse} + {codegen})")?;
22    writeln!(f)
23  })
24}

26/// Formats an elapsed duration cleanly (e.g. `14.3ms`, `450µs`, `1.25s`).
27fn fmt_dur(d: Duration) -> impl fmt::Display {
28  let nanos = d.as_nanos() as u64;
29  let i = nanos.max(1).ilog10().min(9) as usize / 3;
30  let unit = ["n", "µ", "m", ""][i];
31  let time = nanos as f64 / [1e0, 1e3, 1e6, 1e9][i];
32  let prec = i.min(1) * ((time < 99.95) as usize + (time < 9.995) as usize);
33  fmt::from_fn(move |f| write!(f, "{time:.prec$}{unit}s"))
34}

36/// Formats the current UTC system time as `YYYY-MM-DD HH:MM:SS UTC`.
37fn fmt_utc() -> impl fmt::Display {
38  let secs = UNIX_EPOCH.elapsed().map_or(0, |d| d.as_secs());
39  let [mins, sec] = [secs / 60, secs % 60];
40  let [hours, min] = [mins / 60, mins % 60];
41  let [days, hour] = [hours / 24, hours % 24];
42  let (year, month, day) = civil_from_days(days);
43  fmt::from_fn(move |f| write!(f, "{year:04}-{month:02}-{day:02} {hour:02}:{min:02}:{sec:02} UTC"))
44}

46/// Converts days since the Unix epoch (1970-01-01) to a Gregorian `(year, month, day)`.
47///
48/// Implements Howard Hinnant's calendar algorithm for zero-allocation date conversion.
49fn civil_from_days(days: u64) -> (i64, u32, u32) {
50  let z = days as i64 + 719_468;
51  let era = if let 0.. = z { z } else { z - 146_096 } / 146_097;
52  let doe = (z - era * 146_097) as u32;
53  let yoe = (doe - doe / 1_460 + doe / 36_524 - doe / 146_096) / 365;
54  let y = yoe as i64 + era * 400;
55  let doy = doe - (365 * yoe + yoe / 4 - yoe / 100);
56  let mp = (5 * doy + 2) / 153;
57  let d = doy - (153 * mp + 2) / 5 + 1;
58  let (m, y) = if let ..10 = mp { (mp + 3, y) } else { (mp - 9, y + 1) };
59  (y, m, d)
60}
