1//! Conservative detection of recursive Rust layouts in a TL type graph.
2//!
3//! Vertices represent constructor structs and result-category enums. Each
4//! non-unit enum contributes an edge to its constructor payload, and each
5//! constructor contributes edges to directly stored, non-native field types.
6//! `Vec` fields are omitted because their allocation already breaks recursion.
7//!
8//! Every direct field edge inside a strongly connected component is boxed. This
9//! linear-time rule can box more fields than strictly necessary, but guarantees
10//! finite layouts without solving the NP-hard minimum feedback-edge problem.
11
12use td_parser::{Combinator, Definition, DefinitionKind, TypeExpr};
13
14use crate::{graph::Graph, util};
15
16/// Sorted type names and their recursive-layout component IDs.
17#[derive(Debug)]
18pub struct LayoutComponents<'a> {
19  names: Vec<&'a str>,
20  components: Vec<usize>,
21}
22
23impl<'a> LayoutComponents<'a> {
24  /// Builds the direct-storage dependency graph for all type definitions.
25  pub fn from_ast(ast: &[Definition<'a>]) -> Self {
26    let mut names: Vec<_> = ast //.
27      .iter()
28      .filter(|def| def.kind == DefinitionKind::Type)
29      .flat_map(|def| [def.comb.name, def.comb.r#type])
30      .collect();
31    names.sort_unstable();
32    names.dedup();
33
34    let edges = ast
35      .iter()
36      .filter(|def| def.kind == DefinitionKind::Type)
37      .flat_map(|def| {
38        let Combinator { r#type, name, ref fields, .. } = def.comb;
39        let enum_payload = (!fields.is_empty() && r#type != name).then_some([r#type, name]);
40        let direct_fields = fields.iter().filter_map(move |field| Some([name, direct_type(&field.r#type)?]));
41        enum_payload.into_iter().chain(direct_fields)
42      })
43      .filter_map(|edge| match edge.map(|name| names.binary_search(&name)) {
44        [Ok(src), Ok(dst)] => Some([src, dst]),
45        _ => None,
46      })
47      .collect();
48
49    let graph = Graph::from_edges(edges, names.len());
50    let components = graph.strongly_connected_components();
51    Self { names, components }
52  }
53
54  /// Reports whether an existing dependency edge belongs to a layout cycle.
55  pub fn is_recursive_edge(&self, [src, dst]: [&str; 2]) -> bool {
56    let [Ok(src), Ok(dst)] = [src, dst].map(|name| self.names.binary_search(&name)) else { return false };
57    let [Some(&src), Some(&dst)] = [src, dst].map(|idx| self.components.get(idx)) else { return false };
58    src == dst
59  }
60}
61
62/// Returns a directly stored non-native type name.
63fn direct_type<'a>(expr: &TypeExpr<'a>) -> Option<&'a str> {
64  match *expr {
65    TypeExpr::Bare(name) if let None = util::to_native(name) => Some(name),
66    TypeExpr::Bare(_) | TypeExpr::Vector(_) => None,
67  }
68}

70#[cfg(test)]
71mod tests {
72  use td_parser::parse;
73
74  use super::*;
75
76  #[test]
77  fn separates_acyclic_dependencies() {
78    let ast = parse("user id:int32 = User; message author:User = Message;").unwrap();
79    let layouts = LayoutComponents::from_ast(&ast);
80
81    assert!(!layouts.is_recursive_edge(["message", "User"]));
82    assert!(!layouts.is_recursive_edge(["User", "Message"]));
83  }
84
85  #[test]
86  fn detects_multi_type_cycle() {
87    let ast = parse("a next:B = A; b next:C = B; c next:A = C;").unwrap();
88    let layouts = LayoutComponents::from_ast(&ast);
89
90    for edge in [["a", "B"], ["b", "C"], ["c", "A"]] {
91      assert!(layouts.is_recursive_edge(edge));
92    }
93  }
94
95  #[test]
96  fn vector_breaks_cycle() {
97    let ast = parse("folder subfolders:vector<Category> = Category;").unwrap();
98    let layouts = LayoutComponents::from_ast(&ast);
99
100    assert!(!layouts.is_recursive_edge(["folder", "Category"]));
101  }
102}
