9/// Deterministic schema indexes and layout analysis used during generation.
10pub struct Schema<'a> {
11  /// Constructors sorted by `[result_type, constructor_name]`.
12  ctors: Vec<&'a Definition<'a>>,
13  /// Functions retained in schema order.
14  fns: Vec<&'a Definition<'a>>,
15  /// Sorted, deduplicated TL type names for binary search.
16  type_names: Vec<&'a str>,
17  /// Recursive layout cycle detector.
18  layout: Layout<'a>,
19}
20
21impl<'a> Schema<'a> {
22  /// Builds deterministic lookup tables for `ast`.
23  pub fn new(ast: &'a [Definition<'a>]) -> Self {
24    let (mut type_names, mut ctors, mut fns): (Vec<_>, Vec<_>, Vec<_>) = Default::default();
25
26    for def in ast {
27      let target = match def.kind {
28        DefinitionKind::Type => &mut ctors,
29        DefinitionKind::Function => &mut fns,
30      };
31      target.push(def);
32      type_names.push(def.r#type);
33    }
34
35    ctors.sort_unstable_by_key(|def| [def.r#type, def.name]);
36
37    type_names.sort_unstable();
38    type_names.dedup();
39
40    let layout = Layout::new(ast);
41    Self { ctors, fns, type_names, layout }
42  }
43
44  /// Iterates non-primitive TL types and their contiguous constructor groups.
45  pub fn types(&self) -> impl Iterator<Item = (&'a str, &[&'a Definition<'a>])> {
46    self.ctors.chunk_by(|a, b| a.r#type == b.r#type).filter_map(|group| {
47      let &[def, ..] = group else { return None };
48      let None = primitive(def.r#type) else { return None };
49      Some((def.r#type, group))
50    })
51  }
52
53  /// Returns function definitions in their original schema order.
54  pub fn fns(&self) -> &[&'a Definition<'a>] {
55    &self.fns
56  }
57
58  /// Reports whether `name` is a known TL type.
59  pub fn is_type(&self, name: &str) -> bool {
60    self.type_names.binary_search(&name).is_ok()
61  }
62
63  /// Reports whether a reference from `parent` to `target` forms a layout cycle.
64  pub fn is_recursive(&self, parent: &str, target: &str) -> bool {
65    self.layout.is_recursive(parent, target)
66  }
67}

69/// Maps a TL primitive name to its generated Rust type.
70pub fn primitive(name: &str) -> Option<&'static str> {
71  match name {
72    "Int32" | "int32" => Some("i32"),
73    "Int53" | "int53" => Some("i64"),
74    "Int64" | "int64" => Some("i64"),
75    "Double" | "double" => Some("f64"),
76    "String" | "string" => Some("String"),
77    "Vector" | "vector" => Some("Vec<_>"),
78    "Bytes" | "bytes" => Some("Vec<u8>"),
79    "Bool" | "bool" => Some("bool"),
80    _ => None,
81  }
82}

84/// Returns the Serde attribute module path if `expr` requires a custom adapter.
85pub fn serde_with(expr: &TypeExpr) -> Option<&'static str> {
86  match expr {
87    TypeExpr::Bare("bytes") => Some("serde_with::bytes"),
88    TypeExpr::Bare("int64") => Some("serde_with::int64"),
89    TypeExpr::Vector(inner) if matches!(**inner, TypeExpr::Bare("int64")) => Some("serde_with::int64_vec"),
90    _ => None,
91  }
92}

94/// Formats a schema name as a Rust identifier, prefixing `r#` for any keyword.
95pub fn escape_keyword(ident: &str) -> impl fmt::Display {
96  fmt::from_fn(move |f| {
97    if RUST_KEYWORDS.binary_search(&ident).is_ok() {
98      f.write_str("r#")?;
99    }
100    f.write_str(ident)
101  })
102}
