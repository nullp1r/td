1use td_parser::{Combinator, Definition, DefinitionKind};
2
3use crate::scc::SccMap;
4use crate::util;
5
6pub struct Context<'a> {
7  scc: SccMap<'a>,
8  enums: Vec<&'a str>,
9  types: Vec<&'a Combinator<'a>>,
10  fns: Vec<&'a Combinator<'a>>,
11}
12
13impl<'a> Context<'a> {
14  pub fn new(ast: &'a [Definition<'a>]) -> Self {
15    let (mut enums, mut types, mut fns): (Vec<_>, Vec<_>, Vec<_>) = Default::default();
16
17    for d in ast {
18      let combs = match d.kind {
19        DefinitionKind::Type => &mut types,
20        DefinitionKind::Function => &mut fns,
21      };
22      combs.push(&d.comb);
23      enums.push(d.comb.r#type);
24    }
25
26    types.sort_unstable_by_key(|c| [c.r#type, c.name]);
27
28    enums.sort_unstable();
29    enums.dedup();
30
31    let scc = SccMap::from_ast(ast);
32    Self { scc, enums, types, fns }
33  }
34
35  pub fn groups(&self) -> impl Iterator<Item = (&'a str, &[&'a Combinator<'a>])> {
36    let groups = self.types.chunk_by(|a, b| a.r#type == b.r#type);
37    groups.filter_map(|group| {
38      let &[comb, ..] = group else { return None };
39      let None = util::to_native(comb.r#type) else { return None };
40      Some((comb.r#type, group))
41    })
42  }
43
44  pub fn fns(&self) -> &[&'a Combinator<'a>] {
45    &self.fns
46  }
47
48  pub fn is_enum(&self, name: &str) -> bool {
49    self.enums.binary_search(&name).is_ok()
50  }
51
52  pub fn in_same_scc(&self, [a, b]: [&str; 2]) -> bool {
53    self.scc.in_same_scc([a, b])
54  }
55}
