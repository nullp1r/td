1//! Sorted indexes and layout analysis shared by source-formatting helpers.
2
3use td_parser::{Combinator, Definition, DefinitionKind};
4
5use crate::scc::LayoutComponents;
6use crate::util;
7
8/// Deterministic schema indexes used during one generation pass.
9///
10/// Constructor definitions are sorted into contiguous result-type groups. The
11/// separate sorted name table makes enum recognition a binary search, while the
12/// layout components identify conservative indirection points.
13pub struct SchemaIndex<'a> {
14  /// Recursive components in the direct-storage type graph.
15  layouts: LayoutComponents<'a>,
16  /// Sorted, deduplicated TL result-type names.
17  enums: Vec<&'a str>,
18  /// Type constructors sorted by `[result type, constructor name]`.
19  ctors: Vec<&'a Combinator<'a>>,
20  /// Functions retained in schema order.
21  fns: Vec<&'a Combinator<'a>>,
22}
23
24impl<'a> SchemaIndex<'a> {
25  /// Builds deterministic lookup tables for `ast`.
26  pub fn new(ast: &'a [Definition<'a>]) -> Self {
27    let (mut enums, mut ctors, mut fns): (Vec<_>, Vec<_>, Vec<_>) = Default::default();
28
29    for def in ast {
30      let target = match def.kind {
31        DefinitionKind::Type => &mut ctors,
32        DefinitionKind::Function => &mut fns,
33      };
34      target.push(&def.comb);
35      enums.push(def.comb.r#type);
36    }
37
38    ctors.sort_unstable_by_key(|ctor| [ctor.r#type, ctor.name]);
39
40    enums.sort_unstable();
41    enums.dedup();
42
43    let layouts = LayoutComponents::from_ast(ast);
44    Self { layouts, enums, ctors, fns }
45  }
46
47  /// Iterates non-native result types and their contiguous constructor groups.
48  pub fn ctor_groups(&self) -> impl Iterator<Item = (&'a str, &[&'a Combinator<'a>])> {
49    self.ctors.chunk_by(|a, b| a.r#type == b.r#type).filter_map(|group| {
50      let &[comb, ..] = group else { return None };
51      let None = util::to_native(comb.r#type) else { return None };
52      Some((comb.r#type, group))
53    })
54  }
55
56  /// Returns function combinators in their original schema order.
57  pub fn fns(&self) -> &[&'a Combinator<'a>] {
58    &self.fns
59  }
60
61  /// Reports whether `name` is a known TL result category.
62  pub fn is_enum(&self, name: &str) -> bool {
63    self.enums.binary_search(&name).is_ok()
64  }
65
66  /// Reports whether the conservative layout policy boxes this direct field.
67  pub fn needs_box(&self, edge: [&str; 2]) -> bool {
68    self.layouts.is_recursive_edge(edge)
69  }
70}
