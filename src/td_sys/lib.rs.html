1//! Raw C bindings for `TDLib`'s JSON interface.
2//!
3//! These bindings expose `TDLib` API objects as JSON. Object fields keep their
4//! `TDLib` names, and `@type` identifies the object type unless context makes it
5//! unambiguous. Booleans use JSON booleans; `int32`, `int53`, and `double` use
6//! numbers; `int64` and strings use strings; bytes use base64 strings; and
7//! arrays use JSON arrays.
8//!
9//! The main interface is asynchronous. A request may include `@extra`; its
10//! response contains the same value. Every received object contains
11//! `@client_id`, identifying its client.
12//!
13//! Create clients with [`td_create_client_id`], send requests with [`td_send`],
14//! and receive results with [`td_receive`]. Only one thread may receive at a
15//! time, and results must be applied in receive order. Some documented requests
16//! may instead be run synchronously with [`td_execute`]. Clients are destroyed
17//! after they close; close all clients before process exit.
18//!
19//! ```no_run
20//! use td_sys::{td_create_client_id, td_receive, td_send};
21//!
22//! // SAFETY: This call has no pointer arguments.
23//! let client_id = unsafe { td_create_client_id() };
24//! let request = cr#"{"@type":"getOption","name":"version"}"#;
25//! // SAFETY: `client_id` came from TDLib; `request` is static and NUL-terminated.
26//! unsafe { td_send(client_id, request.as_ptr()) };
27//!
28//! loop {
29//!   // SAFETY: This is the program's only receiver thread.
30//!   let response = unsafe { td_receive(10.0) };
31//!   if !response.is_null() {
32//!     // Parse and apply the response before the next receive or execute call.
33//!   }
34//! }
35//! ```
36//!
37//! The `td_json_client_*` functions provide the legacy pointer API, which uses
38//! the same JSON representation and correlation scheme. `TDLib` plans to remove
39//! this API in version 2.0.0. These are raw FFI bindings; callers must uphold
40//! the pointer, callback, threading, and buffer-lifetime contracts documented
41//! on each item.

45use core::ffi::{c_char, c_double, c_int, c_void};

47/// Called when `TDLib` adds a message to its internal log.
48///
49/// `verbosity_level` ranges from -1 through 1024. At 0, `TDLib` crashes after the
50/// callback returns. No `TDLib` function may be called from the callback.
51/// `message` points to a NUL-terminated UTF-8 string valid for the call.
52pub type TdLogMessageCallback = unsafe extern "C" fn(verbosity_level: c_int, message: *const c_char);
53
54/// Called when `TDLib` encounters a fatal error.
55///
56/// No `TDLib` function may be called from the callback. `TDLib` crashes after it
57/// returns. `error_message` points to a NUL-terminated error description.
58pub type TdLogFatalErrorCallback = unsafe extern "C" fn(error_message: *const c_char);
59
60unsafe extern "C" {
61  // --- Modern Client-ID JSON Interface ---
62
63  /// Returns an opaque identifier for a new `TDLib` instance.
64  ///
65  /// The instance sends no updates until its first request. `TDLib` destroys it
66  /// automatically after it is closed. Close every instance before exit.
67  pub fn td_create_client_id() -> c_int;
68
69  /// Sends a request to a `TDLib` client. May be called from any thread.
70  ///
71  /// `client_id` must identify a `TDLib` client. `request` must point to a
72  /// NUL-terminated JSON request valid for the call.
73  pub fn td_send(client_id: c_int, request: *const c_char);
74
75  /// Receives updates and request responses for any client.
76  ///
77  /// May be called from any thread after the first request is sent, but never
78  /// concurrently from two threads. Results must be handled in receive order.
79  ///
80  /// Waits for at most `timeout` seconds and returns null on timeout. Otherwise,
81  /// the returned NUL-terminated JSON string remains valid until the next
82  /// `td_receive` or `td_execute` call.
83  pub fn td_receive(timeout: c_double) -> *const c_char;
84
85  /// Synchronously executes a `TDLib` request. May be called from any thread.
86  ///
87  /// Only requests documented as “Can be called synchronously” are supported.
88  /// `request` must point to a NUL-terminated JSON request valid for the call.
89  /// The returned NUL-terminated JSON string remains valid until the next
90  /// `td_receive` or `td_execute` call.
91  pub fn td_execute(request: *const c_char) -> *const c_char;
92
93  // --- Legacy Pointer-Based JSON Interface (scheduled for removal in TDLib 2.0.0) ---
94
95  /// Creates a `TDLib` instance through the legacy pointer API.
96  pub fn td_json_client_create() -> *mut c_void;
97
98  /// Sends a request through the legacy pointer API. May be called from any thread.
99  ///
100  /// `client` must point to a live `TDLib` instance. `request` must point to a
101  /// NUL-terminated JSON request valid for the call.
102  pub fn td_json_client_send(client: *mut c_void, request: *const c_char);
103
104  /// Receives updates and request responses through the legacy pointer API.
105  ///
106  /// May be called from any thread, but never concurrently from two threads.
107  /// Results must be handled in receive order.
108  ///
109  /// `client` must point to a live `TDLib` instance. The function waits for at
110  /// most `timeout` seconds and returns null on timeout. Otherwise, the returned
111  /// NUL-terminated JSON string remains valid until the next legacy receive or
112  /// execute call on the same thread.
113  pub fn td_json_client_receive(client: *mut c_void, timeout: c_double) -> *const c_char;
114
115  /// Synchronously executes a request through the legacy pointer API.
116  ///
117  /// May be called from any thread, but only a few requests support synchronous
118  /// execution. `client` is currently ignored and may be null. `request` must
119  /// point to a NUL-terminated JSON request valid for the call. The returned
120  /// string remains valid until the next legacy receive or execute call on the
121  /// same thread.
122  pub fn td_json_client_execute(client: *mut c_void, request: *const c_char) -> *const c_char;
123
124  /// Destroys a legacy `TDLib` instance.
125  ///
126  /// `client` must point to a live instance and must not be used after this call.
127  pub fn td_json_client_destroy(client: *mut c_void);
128
129  // --- Logging & Global Configuration ---
130
131  /// Redirects internal logs from the default stream to a file.
132  ///
133  /// `file_path` must point to a NUL-terminated path. An empty path restores the
134  /// default stream. Returns 1 on success or 0 if the file cannot be opened.
135  /// Deprecated upstream in favor of the synchronous `setLogStream` request.
136  pub fn td_set_log_file_path(file_path: *const c_char) -> c_int;
137
138  /// Sets the internal log file size at which `TDLib` rotates the file.
139  ///
140  /// The size must be positive. It defaults to 10 MB and is unused when logging
141  /// to the default stream. Deprecated upstream in favor of `setLogStream`.
142  pub fn td_set_log_max_file_size(max_file_size: i64);
143
144  /// Sets the internal log verbosity level, which defaults to 5.
145  ///
146  /// Levels 0 through 5 select fatal, error, warning/debug-warning,
147  /// informational, debug, and verbose-debug logging. Levels through 1024
148  /// enable more logging. Deprecated upstream in favor of the synchronous
149  /// `setLogVerbosityLevel` request.
150  pub fn td_set_log_verbosity_level(new_verbosity_level: c_int);
151
152  /// Sets the callback for messages up to `max_verbosity_level`.
153  ///
154  /// No `TDLib` function may be called from the callback. Pass `None` to remove it.
155  pub fn td_set_log_message_callback(max_verbosity_level: c_int, callback: Option<TdLogMessageCallback>);
156
157  /// Sets the callback for fatal errors. Pass `None` to remove it.
158  ///
159  /// No `TDLib` function may be called from the callback. `TDLib` crashes after the
160  /// callback returns. Deprecated upstream in favor of
161  /// `td_set_log_message_callback`.
162  pub fn td_set_log_fatal_error_callback(callback: Option<TdLogFatalErrorCallback>);
163}

165#[cfg(test)]
166mod tests {
167  use super::*;
168  use core::ffi::CStr;
169  use std::sync::{Mutex, Once};
170
171  static CALLS: Mutex<()> = Mutex::new(());
172  static SILENCE: Once = Once::new();
173
174  fn silence_logs() {
175    // SAFETY: The call passes no pointers or borrowed storage.
176    SILENCE.call_once(|| unsafe { td_set_log_verbosity_level(0) });
177  }
178
179  #[test]
180  fn stateless_execute() {
181    silence_logs();
182    let _calls = CALLS.lock().expect("lock poisoned");
183
184    let req = cr#"{"@type":"getTextEntities","text":"@telegram https://t.me"}"#;
185    // SAFETY: `req` is a static, NUL-terminated C string.
186    let res_ptr = unsafe { td_execute(req.as_ptr()) };
187    assert!(!res_ptr.is_null(), "td_execute response should not be null");
188
189    // SAFETY: `res_ptr` is a non-null, NUL-terminated TDLib response.
190    // `CALLS` keeps its buffer valid while borrowed.
191    let res_str = unsafe { CStr::from_ptr(res_ptr) }.to_str().expect("valid utf-8 response");
192    assert!(res_str.contains(r#""offset":0,"length":9,"type":{"@type":"textEntityTypeMention"}"#));
193    assert!(res_str.contains(r#""offset":10,"length":12,"type":{"@type":"textEntityTypeUrl"}"#));
194  }
195
196  #[test]
197  fn client_id_lifecycle() {
198    silence_logs();
199
200    // SAFETY: The call takes no arguments and returns an opaque ID by value.
201    let client_id = unsafe { td_create_client_id() };
202    assert!(client_id > 0, "client ID should be positive");
203
204    let _calls = CALLS.lock().expect("lock poisoned");
205    let req = cr#"{"@type":"getOption","name":"version"}"#;
206    // SAFETY: `client_id` came from TDLib; `req` is static and NUL-terminated.
207    unsafe { td_send(client_id, req.as_ptr()) }
208
209    // SAFETY: `CALLS` serializes all `td_receive` and `td_execute` calls.
210    let res_ptr = unsafe { td_receive(1.0) };
211    if !res_ptr.is_null() {
212      // SAFETY: `res_ptr` is a non-null, NUL-terminated TDLib response.
213      // `CALLS` keeps its buffer valid while borrowed.
214      let res_str = unsafe { CStr::from_ptr(res_ptr) }.to_str().expect("valid utf-8 response");
215      assert!(!res_str.is_empty());
216    }
217  }
218
219  #[test]
220  fn legacy_pointer_lifecycle() {
221    silence_logs();
222
223    // SAFETY: The call takes no arguments; Rust treats the returned pointer as opaque.
224    let client = unsafe { td_json_client_create() };
225    assert!(!client.is_null(), "client pointer should not be null");
226
227    // SAFETY: `client` is non-null, live, and not used after this call.
228    unsafe { td_json_client_destroy(client) }
229  }
