25pub struct PendingMessages {

43impl Connection {
44  pub async fn messages<F: Function>(&self, request: &F, progress: bool) -> Result<PendingMessages> {
45    let (reply, response) = oneshot::channel();
46    self.submit(request, PendingReply::Messages { progress, reply })?;
47    response.await.map_err(|_| Error::Disconnected)?
48  }
49
50  pub fn observe_download(&self, request: &fns::downloadFile) -> watch::Receiver<Sample> {
51    // Register before submission, including for immediately cached downloads.
52    let (samples, receiver) = watch::channel((0, Progress::default()));
53    let scope = Scope::Download { offset: request.offset, limit: request.limit };
54    let mut registry = self.registry.lock().unwrap();
55    registry.prune_observers();
56    registry.file_observers.push(Observation { file_id: request.file_id, scope, samples });
57    receiver
58  }
59}

61impl PendingMessages {
62  pub async fn finish(
63    self,
64    connection: &Connection,
65    cancel: Option<&CancellationToken>,
66    callback: Option<&mut (dyn FnMut(usize, Progress) + Send)>,
67  ) -> Vec<Result<types::message>> {
68    // Waiters are already bound. Sequential awaits order results, not native
69    // sends; with_progress stays driven during cancellation cleanup as well.
70    let completion = async {
71      let mut results = Vec::with_capacity(self.pending.len());
72      for message in self.pending {
73        results.push(message.finish(connection, cancel).await);
74      }
75      results
76    };
77    with_progress(completion, self.samples, callback).await
78  }
79}

81impl PendingMessage {
82  pub async fn finish(mut self, connection: &Connection, cancel: Option<&CancellationToken>) -> Result<types::message> {
83    tokio::select! {
84      biased;
85      result = &mut self.terminal => result.unwrap_or(Err(Error::Disconnected)),
86      () = cancelled(cancel) => self.cancel(connection).await,
87    }
88  }
89
90  async fn cancel(self, connection: &Connection) -> Result<types::message> {
91    let Self { key, terminal } = self;
92    let pending = || connection.registry.lock().unwrap().pending_messages.contains_key(&key);
93    // Settlement removes the key before publishing its result. This prevents
94    // knowingly deleting a completed send, not a server-side acceptance race.
95    if pending() {
96      let request = fns::deleteMessages { chat_id: key.chat_id, message_ids: vec![key.message_id], revoke: true };
97      let deletion = connection.request(&request).await;
98      // A terminal result wins if it arrived during deletion. Never delete a final ID.
99      if let Err(error) = deletion
100        && pending()
101      {
102        return Err(error);
103      }
104    }
105    match terminal.await.map_err(|_| Error::Disconnected)? {
106      Err(Error::MessageDeleted(_)) => Err(Error::Cancelled),
107      result => result,
108    }
109  }
110}

112impl Registry {
113  pub fn bind(&mut self, messages: impl IntoIterator<Item = types::message>, progress: bool) -> PendingMessages {
114    let messages = messages.into_iter();
115    let mut pending = Vec::with_capacity(messages.size_hint().0);
116    let mut samples = None;
117    if progress {
118      self.prune_observers();
119    }
120    for (index, message) in messages.enumerate() {
121      let key = MessageKey { chat_id: message.chat_id, message_id: message.id };
122      if progress && let Some(file) = primary_file(&message) {
123        let (sender, _) = samples.get_or_insert_with(|| watch::channel((0, Progress::default())));
124        self.file_observers.push(Observation { file_id: file.id, scope: Scope::Upload(index), samples: sender.clone() });
125      }
126      let (reply, terminal) = oneshot::channel();
127      // Live temporary keys are unique. Abandoned waiters remain until terminal
128      // updates or teardown. Already-final responses share the completion path.
129      if let Some(MessageSendingState::messageSendingStatePending(_)) = message.sending_state {
130        self.pending_messages.insert(key, reply);
131      } else {
132        let _ = reply.send(Ok(message));
133      }
134      pending.push(PendingMessage { key, terminal });
135    }
136    PendingMessages { pending, samples: samples.map(|(_, receiver)| receiver) }
137  }
138
139  fn prune_observers(&mut self) {
140    // Registration also prunes files that never emit another update after the
141    // operation drops. Dropping observation performs no native cancellation.
142    self.file_observers.retain(|observer| observer.samples.receiver_count() != 0);
143  }
144
145  pub fn observe_file(&mut self, file: &types::file) {
146    // ponytail: O(active observers), contiguous and allocation-light for small sets.
147    // Restore keyed routing if deployments need many simultaneous transfers.
148    self.file_observers.retain(|observer| {
149      if observer.file_id != file.id {
150        return observer.samples.receiver_count() != 0;
151      }
152      let sample = match observer.scope {
153        Scope::Upload(index) => (index, Progress { current: file.remote.uploaded_size, total: file.size.max(file.expected_size) }),
154        Scope::Download { offset, limit } => (0, Progress::download(file, offset, limit)),
155      };
156      observer.samples.send(sample).is_ok()
157    });
158  }
159
160  pub fn observe_message(&mut self, update: &Update) {
161    match update {
162      Update::updateMessageSendSucceeded(update) => {
163        let key = MessageKey { chat_id: update.message.chat_id, message_id: update.old_message_id };
164        self.settle(key, || Ok(update.message.clone()));
165      }
166      Update::updateMessageSendFailed(update) => {
167        let key = MessageKey { chat_id: update.message.chat_id, message_id: update.old_message_id };
168        self.settle(key, || Err(Error::MessageFailed(key, update.error.clone())));
169      }
170      Update::updateDeleteMessages(update) if !update.from_cache => {
171        for &message_id in &update.message_ids {
172          let key = MessageKey { chat_id: update.chat_id, message_id };
173          self.settle(key, || Err(Error::MessageDeleted(key)));
174        }
175      }
176      _ => {}
177    }
178  }
179
180  fn settle(&mut self, key: MessageKey, result: impl FnOnce() -> Result<types::message>) {
181    // Clone only for live waiters: the original update goes to the application.
182    if let Some(reply) = self.pending_messages.remove(&key)
183      && !reply.is_closed()
184    {
185      let _ = reply.send(result());
186    }
187  }
188}

190impl Progress {
191  fn download(file: &types::file, offset: i64, limit: i64) -> Self {
192    let local = &file.local;
193    let available = (file.size.max(file.expected_size) - offset).max(0);
194    let total = if limit > 0 { available.min(limit) } else { available };
195    let prefix_end = local.download_offset + local.downloaded_prefix_size;
196    // Bytes beyond a missing prefix do not count. Range arithmetic must fit i64;
197    // clamping describes overlap, not sanitization of invalid request arguments.
198    let current = if local.download_offset <= offset { (prefix_end - offset).max(0) } else { 0 };
199    // Unknown size stays total=0, but an explicit limit still bounds current.
200    let bound = match (total, limit) {
201      (1.., _) => total,
202      (_, 1..) => limit,
203      _ => current,
204    };
205    Self { current: current.min(bound), total }
206  }
207}

209pub async fn with_progress<T>(
210  completion: impl Future<Output = T>,
211  samples: Option<watch::Receiver<Sample>>,
212  callback: Option<&mut (dyn FnMut(usize, Progress) + Send)>,
213) -> T {
214  tokio::pin!(completion);
215  if let (Some(mut samples), Some(callback)) = (samples, callback) {
216    loop {
217      tokio::select! {
218        biased;
219        // Completion wins; the callback contract promises no final-sample flush.
220        result = &mut completion => return result,
221        changed = samples.changed() => {
222          let Ok(()) = changed else { break };
223          // Release the watch borrow before invoking application code.
224          let (index, progress) = *samples.borrow_and_update();
225          callback(index, progress);
226        }
227      }
228    }
229  }
230  // A closed observer is not completion. Do not spin on its ready error branch.
231  completion.await
232}

234pub async fn cancelled(token: Option<&CancellationToken>) {
235  match token {
236    Some(token) => token.cancelled().await,
237    None => future::pending().await,
238  }

241fn primary_file(message: &types::message) -> Option<&types::file> {
242  // Track the primary payload, not thumbnails or a recursive walk of generated types.
243  match &message.content {
244    MessageContent::messageAnimation(content) => Some(&content.animation.animation),
245    MessageContent::messageAudio(content) => Some(&content.audio.audio),
246    MessageContent::messageDocument(content) => Some(&content.document.document),
247    MessageContent::messagePhoto(content) => content.photo.sizes.last().map(|size| &size.photo),
248    MessageContent::messageSticker(content) => Some(&content.sticker.sticker),
249    MessageContent::messageVideo(content) => Some(&content.video.video),
250    MessageContent::messageVideoNote(content) => Some(&content.video_note.video),
251    MessageContent::messageVoiceNote(content) => Some(&content.voice_note.voice),
252    _ => None,
253  }
254}

256pub fn parse_messages(raw: &[u8], kind: &str) -> Result<impl Iterator<Item = types::message>> {
257  // Keep the singleton inline instead of allocating a temporary binding vector.
258  let (single, batch) = match kind {
259    "message" => {
260      let Message::message(message) = serde_json::from_slice(raw)?;
261      (Some(message), Vec::new())
262    }
263    "messages" => {
264      let Messages::messages(messages) = serde_json::from_slice(raw)?;
265      (None, messages.messages.ok_or(Error::UnexpectedResponse("batch response omitted messages"))?)
266    }
267    _ => return Err(Error::UnexpectedResponse("tracked send returned an unexpected type")),
268  };
269  Ok(single.into_iter().chain(batch))
270}

333      let error = types::error { code: 400, message: "failed".into() };

342    let [first, second, third]: [Result<types::message>; 3] = results.try_into().unwrap();
343    assert_matches!(first, Err(Error::MessageFailed(MessageKey { message_id: 10, .. }, error)) if error.code == 400);
344    assert_matches!(second, Ok(types::message { id: 21, .. }));
345    assert_matches!(third, Err(Error::MessageDeleted(MessageKey { message_id: 12, .. })));
346  }

441  fn observers_coalesce_by_operation_and_prune_abandoned_waits() {
442    let (connection, _updates) = Connection::fixture();
443    let (upload, mut uploads) = watch::channel((0, Progress::default()));
444    let request = fns::downloadFile { file_id: 7, offset: 20, limit: 50, ..Default::default() };
445    let mut downloads = connection.observe_download(&request);
446    let mut registry = connection.registry.lock().unwrap();
447    registry.file_observers.push(Observation { file_id: 7, scope: Scope::Upload(2), samples: upload.clone() });
448    registry.file_observers.push(Observation { file_id: 8, scope: Scope::Upload(3), samples: upload });
449    let remote = types::remoteFile { uploaded_size: 60, ..Default::default() };
450    let local = types::localFile { downloaded_prefix_size: 100, ..Default::default() };
451    let file = types::file { id: 7, size: 100, local, remote, ..Default::default() };
452    registry.observe_file(&file);
453    assert_eq!(*uploads.borrow_and_update(), (2, Progress { current: 60, total: 100 }));
454    assert_eq!(*downloads.borrow_and_update(), (0, Progress { current: 50, total: 50 }));
455
456    registry.observe_file(&file);
457    registry.observe_file(&types::file { id: 8, ..file });
458    assert_eq!(*uploads.borrow_and_update(), (3, Progress { current: 60, total: 100 }));
459    drop((uploads, downloads));
460    registry.observe_file(&types::file::default());
461    assert!(registry.file_observers.is_empty());
462  }
