1//! Borrowed syntax tree for the `TDLib` API schema.
2
3/// Whether a schema definition declares an object constructor or a function.
4#[derive(Debug, PartialEq, Eq, Clone, Copy)]
5pub enum DefinitionKind {
6  /// A constructor in the schema's types section.
7  Type,
8  /// A request in the schema's functions section.
9  Function,
10}
11
12/// One classified constructor or function definition.
13///
14/// Every borrowed string in the contained combinator points into the original
15/// schema passed to [`crate::parse`].
16#[derive(Debug, PartialEq, Eq, Clone)]
17pub struct Definition<'a> {
18  /// Section in which the definition appeared.
19  pub kind: DefinitionKind,
20  /// Shared constructor/function syntax parsed from the definition.
21  pub comb: Combinator<'a>,
22}
23
24/// The common shape of a TL object constructor or function.
25#[derive(Debug, PartialEq, Eq, Clone)]
26pub struct Combinator<'a> {
27  /// Result category for a constructor, or response type for a function.
28  pub r#type: &'a str,
29  /// Constructor or function name used as the JSON `@type` value.
30  pub name: &'a str,
31  /// Parameters in schema order.
32  pub fields: Vec<Field<'a>>,
33  /// Text from the definition's `@description` tag, if present.
34  pub desc: Option<&'a str>,
35  /// Description from a preceding `@class` declaration, if present.
36  ///
37  /// Class metadata describes the whole result category rather than this one
38  /// constructor. The generator locates it while grouping constructors.
39  pub meta: Option<&'a str>,
40}
41
42/// One named constructor field or function parameter.
43#[derive(Debug, PartialEq, Eq, Clone)]
44pub struct Field<'a> {
45  /// Whether the schema prose permits null for this field.
46  ///
47  /// TL does not encode JSON nullability in the field type, so this is derived
48  /// from the upstream phrases “may be null” and “pass null”.
49  pub is_optional: bool,
50  /// Field name used on the JSON wire.
51  pub name: &'a str,
52  /// Parsed field type.
53  pub r#type: TypeExpr<'a>,
54  /// Text from the field's documentation tag, if present.
55  pub desc: Option<&'a str>,
56}
57
58/// A field type relevant to generated Rust storage.
59#[derive(Debug, PartialEq, Eq, Clone)]
60pub enum TypeExpr<'a> {
61  /// A primitive, constructor, or result-category name.
62  Bare(&'a str),
63  /// A vector whose element type is another expression.
64  Vector(Box<TypeExpr<'a>>),
65}
