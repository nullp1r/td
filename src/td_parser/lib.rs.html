1//! Zero-copy parsing for `TDLib`'s Type Language API schema.
2//!
3//! This crate parses the subset of TL used by `td_api.tl` into a compact,
4//! borrowed syntax tree. Constructor and function names, type names, and
5//! documentation remain slices of the input; only the definition and field
6//! structure is allocated. The parser also carries `//@description`,
7//! `//@class`, and field documentation into the AST for code generation.
8//!
9//! The representation is intentionally specialized for generating the JSON API:
10//! constructor IDs and generic declarations are accepted but discarded, while
11//! vectors and direct named references are retained because they determine the
12//! generated Rust layout.
13//!
14//! ```
15//! use td_parser::{DefinitionKind, TypeExpr, parse};
16//!
17//! let schema = r#"
18//!   //@description A user
19//!   //@id User identifier
20//!   user id:int64 = User;
21//!   ---functions---
22//!   //@description Returns a user
23//!   //@id User identifier
24//!   getUser id:int64 = User;
25//! "#;
26//!
27//! let definitions = parse(schema).expect("valid TD API schema");
28//! assert_eq!(definitions.len(), 2);
29//! assert_eq!(definitions[0].kind, DefinitionKind::Type);
30//! assert_eq!(definitions[0].comb.fields[0].r#type, TypeExpr::Bare("int64"));
31//! assert_eq!(definitions[1].kind, DefinitionKind::Function);
32//! ```
