1//! Recursive-descent parsing for the JSON-oriented subset of TL.
2//!
3//! The cursor advances through the source once. Balanced generic and parameter
4//! groups are skipped because their contents do not affect JSON code generation;
5//! direct named references and vector nesting are retained in [`TypeExpr`].
6
7use std::iter;
8
9use crate::ast::{Combinator, Definition, DefinitionKind, Field, TypeExpr};
10use crate::cursor::Cursor;
11use crate::error::Error;
12
13impl<'a> Cursor<'a> {
14  /// Skips constructor IDs and TL generic/parameter declarations after a name.
15  fn skip_params(&mut self) -> Result<(), Error<'a>> {
16    loop {
17      self.skip_ws();
18      if self.maybe("#") {
19        self.hex();
20        self.skip_ws();
21        self.maybe_balanced(['[', ']'])?;
22      } else if !self.maybe_balanced(['{', '}'])? && !self.maybe("?") {
23        return Ok(());
24      }
25    }
26  }
27
28  /// Parses one field and attaches its same-named documentation tag.
29  fn field(&mut self, doc: &'a str) -> Result<Field<'a>, Error<'a>> {
30    let name = self.ident().ok_or(Error::UnexpectedInput(self.rest))?;
31    self.skip_ws();
32    self.expect(":")?;
33    let r#type = self.type_expr()?;
34    let desc = doc_tags(doc).find(|&[n, _]| n == name).map(|[_, d]| d);
35    let is_optional = desc.is_some_and(is_optional);
36    Ok(Field { is_optional, name, r#type, desc })
37  }
38
39  /// Parses the storage-relevant part of a TL type expression.
40  fn type_expr(&mut self) -> Result<TypeExpr<'a>, Error<'a>> {
41    self.skip_ws();
42    if self.maybe("vector") {
43      self.skip_ws();
44      if self.maybe("<") {
45        let inner = self.type_expr()?;
46        self.skip_ws();
47        self.expect(">")?;
48        return Ok(TypeExpr::Vector(Box::new(inner)));
49      }
50      return Ok(TypeExpr::Bare("vector"));
51    }
52    if self.maybe("?") {
53      return Ok(TypeExpr::Bare("?"));
54    }
55    if self.maybe("#") {
56      return Ok(TypeExpr::Bare("#"));
57    }
58    let Some(name) = self.ident() else {
59      return Err(Error::ExpectedTypeExpr);
60    };
61    self.skip_ws();
62    self.maybe_balanced(['<', '>'])?;
63    Ok(TypeExpr::Bare(name))
64  }
65
66  /// Parses the next definition, updating the active section at marker lines.
67  fn definition(&mut self, kind: &mut DefinitionKind) -> Result<Option<Definition<'a>>, Error<'a>> {
68    loop {
69      let doc = self.comments();
70
71      if self.rest.is_empty() {
72        return Ok(None);
73      }
74
75      if self.maybe("---") {
76        let (marker, tail) = self.rest.split_once('\n').unwrap_or((self.rest, ""));
77        self.rest = tail;
78        *kind = match marker.trim_matches(['-', ' ']) {
79          "types" => DefinitionKind::Type,
80          "functions" => DefinitionKind::Function,
81          _ => return Err(Error::ExpectedDefinitionKind),
82        };
83        continue;
84      }
85
86      let Some(name) = self.ident() else {
87        return Err(Error::UnexpectedInput(self.rest));
88      };
89      self.skip_params()?;
90
91      let mut fields = Vec::new();
92      while {
93        self.skip_ws();
94        !self.maybe("=")
95      } {
96        fields.push(self.field(doc)?);
97      }
98
99      self.skip_ws();
100      let Some(r#type) = self.ident() else {
101        return Err(Error::ExpectedEnum);
102      };
103      self.skip_ws();
104      while let Some(_) = self.ident() {
105        self.skip_ws();
106      }
107      self.expect(";")?;
108
109      let [desc, meta] = desc_and_meta_desc(doc);
110      let comb = Combinator { r#type, name, fields, desc, meta };
111      return Ok(Some(Definition { kind: *kind, comb }));
112    }
113  }
114}
115
116fn is_optional(desc: &str) -> bool {
117  desc.contains("may be null") || desc.contains("pass null")
118}
119
120/// Separates constructor documentation from category-level `@class` metadata.
121fn desc_and_meta_desc(doc: &str) -> [Option<&str>; 2] {
122  let (mut is_meta, mut meta, mut desc) = Default::default();
123  for [key, value] in doc_tags(doc) {
124    match key {
125      "class" => is_meta = true,
126      "description" if is_meta => (is_meta, meta) = (false, Some(value)),
127      "description" => desc = Some(value),
128      _ => {}
129    }
130  }
131  [desc, meta]
132}

134/// Iterates both line-leading and compact inline `@name value` documentation tags.
135///
136/// Upstream uses both `//@description ...\n//@field ...` and the compact
137/// `//@description ... @field ...` form. A space before `@` is the delimiter for
138/// an inline tag; continuation lines beginning `//-` remain part of its value.
139fn doc_tags(doc: &str) -> impl Iterator<Item = [&str; 2]> {
140  doc.split("//@").skip(1).flat_map(|part| part.split(" @")).filter_map(|part| {
141    let (key, value) = part.trim_ascii().split_once(char::is_whitespace)?;
142    Some([key, value])
143  })
144}

146/// Parses a `TDLib` API schema into definitions borrowing from `input`.
147///
148/// The parser recognizes type/function section markers, constructors, functions,
149/// direct and vector field types, and the documentation syntax used by
150/// `td_api.tl`. Constructor IDs and generic declarations are validated only for
151/// balanced delimiters, then omitted from the returned JSON-oriented AST.
152/// Definitions before the first section marker are treated as types.
153///
154/// # Errors
155///
156/// Returns the first malformed delimiter, missing token, invalid section marker,
157/// or unexpected input suffix.
158pub fn parse(input: &str) -> Result<Vec<Definition<'_>>, Error<'_>> {
159  let mut cur = Cursor::new(input);
160  let mut kind = DefinitionKind::Type;
161  iter::from_fn(|| cur.definition(&mut kind).transpose()).collect()
162}
